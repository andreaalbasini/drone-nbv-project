In another terminal verify which dictionary it is:

python3 -c "
import cv2
import numpy as np

frame = cv2.imread('/tmp/test_aruco.jpg')
gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

dictionaries = {
    '4X4_50': cv2.aruco.DICT_4X4_50,
    '4X4_100': cv2.aruco.DICT_4X4_100,
    '5X5_50': cv2.aruco.DICT_5X5_50,
    '6X6_50': cv2.aruco.DICT_6X6_50,
    'ARUCO_ORIGINAL': cv2.aruco.DICT_ARUCO_ORIGINAL,
}

params = cv2.aruco.DetectorParameters_create()

for name, d in dictionaries.items():
    aruco_dict = cv2.aruco.getPredefinedDictionary(d)
    corners, ids, _ = cv2.aruco.detectMarkers(gray, aruco_dict, parameters=params)
    if ids is not None:
        print(f'FOUND with {name}: ids={ids.flatten().tolist()}')
    else:
        print(f'Not found with {name}')
"
